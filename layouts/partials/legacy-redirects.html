{{- /* Writes a redirect page at each old address in data/legacy_redirects.yaml: the old /tag/ pages (the site uses /categories/ now) and three renamed categories. They used to survive only because every deploy kept the previous build's files. Same page Hugo writes for `aliases:` (canonical link plus instant refresh), so browsers and search engines both follow it. Called once, from the home page. */ -}} {{- range $from, $to := site.Data.legacy_redirects }} {{- $url := $to | absURL }} {{- $page := printf "\n\n \n %s\n \n \n \n \n\n" $url $url $url }} {{- (resources.FromString (printf "%sindex.html" (strings.TrimPrefix "/" $from)) $page).Publish }} {{- end }}
